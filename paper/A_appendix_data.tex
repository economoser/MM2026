\begin{figure}[htbp!]
    \subfloat[Men]{\includegraphics[width=0.50\columnwidth]{figures/figureA1a.eps}\label{fig:exp_act_pot_A}}% , trim = 0 0 0 23, clip
    \subfloat[Women]{\includegraphics[width=0.50\columnwidth]{figures/figureA1b.eps}\label{fig:exp_act_pot_B}}
    %
    \caption{\label{fig:exp_act_pot}Percentiles of actual experience conditional on potential experience}
    %
    \begin{figurenotes}
        %
        \footnotesize
        %
        Figure shows percentiles of actual against potential experience separately for men (Panel \subref{fig:exp_act_pot_A}) and women (Panel \subref{fig:exp_act_pot_B}). Actual experience is constructed from panel data for 1985--\yearmax{}, while $\text{potential experience} = \text{age} - \text{years of education} + 6$. Solid line represents the 45-degree line, for which actual experience equals potential experience. %
    \end{figurenotes}
